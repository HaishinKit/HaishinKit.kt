cmake_minimum_required(VERSION 3.22.1)
project(haishinkit_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
    add_library(haishinkit_audio SHARED audio_mixer.cpp audio_jni.cpp)
    target_compile_options(haishinkit_audio PRIVATE -O3 -ffp-contract=off -Wall -Wextra -Werror)
    target_link_options(haishinkit_audio PRIVATE "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    add_executable(audio_mixer_test audio_mixer.cpp audio_mixer_test.cpp)
    target_compile_options(audio_mixer_test PRIVATE -O3 -ffp-contract=off -Wall -Wextra -Werror)
    add_test(NAME audio_mixer_test COMMAND audio_mixer_test)
endif()
